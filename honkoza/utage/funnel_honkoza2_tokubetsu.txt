  function track(name){try{if(window.fbq)fbq('trackCustom',name,{ref:ref,page:'tokubetsu_498'});}catch(e){}}
  var go=function(e){e.preventDefault();track('ToApply');document.getElementById('fx-formhead').scrollIntoView({behavior:RM?'auto':'smooth',block:'start'});};
  [].slice.call(document.querySelectorAll('[data-go]')).forEach(function(a){a.addEventListener('click',go);});
  var cs=document.querySelector('[data-consult]');if(cs)cs.addEventListener('click',function(){track('ToConsult');});

  /* 下に固定のボタン（スマホ） */
  var sk=null;
  if(stage!=='closed'){
    sk=document.createElement('div');sk.id='fx-stick';
    sk.innerHTML=(target?'<span class="mini" id="fx-mini"></span>':'')+'<a class="fx-cta" href="#fx-formhead">この参加費で申し込む</a>';
    document.body.appendChild(sk);sk.querySelector('a').addEventListener('click',go);
  }

  /* 残り時間（毎秒） */
  if(target){
    var els=[].slice.call(document.querySelectorAll('#fx-cd [data-u]')),mini=document.getElementById('fx-mini');
    var tick=function(){var p=parts(msLeft(target));var v=[String(p[0]),pad(p[1]),pad(p[2]),pad(p[3])];
      els.forEach(function(e,i){if(e.textContent!==v[i]){e.textContent=v[i];if(!RM){e.classList.remove('tick');void e.offsetWidth;e.classList.add('tick');}}});
      if(mini)mini.textContent='締切まで あと'+p[0]+'日 '+pad(p[1])+':'+pad(p[2])+':'+pad(p[3]);};
    tick();setInterval(tick,1000);
  }

  /* 上の進みバー */
  var pg=document.createElement('div');pg.id='fx-prog';document.body.appendChild(pg);
  var onScroll=function(){var h=document.documentElement;var r=h.scrollTop/((h.scrollHeight-h.clientHeight)||1);pg.style.width=(Math.min(1,Math.max(0,r))*100)+'%';};
  window.addEventListener('scroll',onScroll,{passive:true});onScroll();

  /* 招待状が開く・数字が回る */
  var card=document.getElementById('fx-card');
  var rollIt=function(){[].slice.call(card.querySelectorAll('.roll')).forEach(function(r,i){var d=+r.getAttribute('data-d');var inner=r.firstChild;inner.style.transitionDelay=(RM?0:.35+i*.09)+'s';inner.style.transform='translateY(-'+d+'em)';});};
  if(RM||!card.animate){rollIt();}
  else{
    card.animate([{opacity:0,transform:'perspective(1400px) rotateX(24deg) translateY(70px) scale(.94)'},{opacity:1,transform:'perspective(1400px) rotateX(0) translateY(0) scale(1)'}],{duration:1300,easing:'cubic-bezier(.16,1,.3,1)',fill:'both'});
    document.getElementById('fx-wax').animate([{opacity:0,transform:'scale(2.2) rotate(-25deg)'},{opacity:1,transform:'scale(.92) rotate(0)'},{opacity:1,transform:'scale(1) rotate(0)'}],{duration:700,delay:1000,easing:'cubic-bezier(.3,1.4,.5,1)',fill:'both'});
    [].slice.call(document.querySelectorAll('#fx-hero .orn,#fx-cd,#fx-hero .fx-cta,#fx-hero .fx-cta-sub')).forEach(function(e,i){e.animate([{opacity:0,transform:'translateY(14px)'},{opacity:1,transform:'none'}],{duration:800,delay:900+i*180,easing:'ease-out',fill:'both'});});
    setTimeout(rollIt,700);
  }

  /* 金の粒（ヒーローだけ・見えているときだけ動かす） */
  var cv=document.getElementById('fx-dust');
  if(cv&&!RM&&cv.getContext){
    var cx=cv.getContext('2d'),hero=document.getElementById('fx-hero'),W,H,dpr=Math.min(2,window.devicePixelRatio||1),P=[],vis=true;
    var size=function(){W=hero.clientWidth;H=hero.clientHeight;cv.width=W*dpr;cv.height=H*dpr;cx.setTransform(dpr,0,0,dpr,0,0);};
    size();window.addEventListener('resize',size);
    var N=W<500?34:58;
    for(var i=0;i<N;i++)P.push({x:Math.random()*W,y:Math.random()*H,r:Math.random()*1.6+.4,s:Math.random()*.35+.1,a:Math.random()*Math.PI*2,w:Math.random()*.02+.006});
    var draw=function(){if(vis){cx.clearRect(0,0,W,H);P.forEach(function(p){p.y-=p.s;p.a+=p.w;p.x+=Math.sin(p.a)*.25;if(p.y<-6){p.y=H+6;p.x=Math.random()*W;}
      var al=.35+.45*Math.abs(Math.sin(p.a*1.7));cx.beginPath();cx.arc(p.x,p.y,p.r,0,7);cx.fillStyle='rgba(231,207,142,'+al+')';cx.shadowColor='rgba(231,207,142,.9)';cx.shadowBlur=6;cx.fill();});}
      requestAnimationFrame(draw);};
    if('IntersectionObserver' in window)new IntersectionObserver(function(es){vis=es[0].isIntersecting;}).observe(hero);
    draw();
  }

  /* 出てくる動き */
  if('IntersectionObserver' in window&&!RM){
    var io=new IntersectionObserver(function(es){es.forEach(function(x){if(x.isIntersecting){x.target.classList.add('on');io.unobserve(x.target);}});},{threshold:.12,rootMargin:'0px 0px -6% 0px'});
    [].slice.call(document.querySelectorAll('.fx-rv,#fx-letter')).forEach(function(el){io.observe(el);});
    if(sk){
      var heroOut=false,formIn=false;
      var upd=function(){sk.classList.toggle('on',heroOut&&!formIn);};
      new IntersectionObserver(function(es){heroOut=!es[0].isIntersecting;upd();}).observe(document.getElementById('fx-hero'));
      new IntersectionObserver(function(es){formIn=es[0].isIntersecting||es[0].boundingClientRect.top<0;upd();},{threshold:0}).observe(document.getElementById('fx-formhead'));
    }
  }else{
    [].slice.call(document.querySelectorAll('.fx-rv,#fx-letter')).forEach(function(el){el.classList.add('on');});
    if(sk){window.addEventListener('scroll',function(){var f=document.getElementById('fx-formhead').getBoundingClientRect().top;sk.classList.toggle('on',window.scrollY>500&&f>window.innerHeight);},{passive:true});}
  }
}
if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',run);else run();
})();
